추천 Apps Script 코드 (Google 스프레드시트 자동 기록 및 조회) 
// Google Apps Script (Code.gs)
const VALID_API_KEY = "YOUR_CUSTOM_API_KEY"; // 관리자 콘솔의 인증 API Key와 동일하게 설정

function doPost(e) {
  try {
    const contents = e.postData ? e.postData.contents : "{}";
    const payload = JSON.parse(contents);

    // API Key 검증
    if (payload.apiKey !== VALID_API_KEY) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Invalid API Key" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. 요약 시트 업데이트
    let summarySheet = ss.getSheetByName("요약통계");
    if (!summarySheet) {
      summarySheet = ss.insertSheet("요약통계");
      summarySheet.appendRow(["항목", "값", "마지막 갱신 일시"]);
    }
    const sm = payload.analytics?.summary || {};
    summarySheet.clearContents();
    summarySheet.appendRow(["항목", "값", "마지막 갱신 일시"]);
    summarySheet.appendRow(["총 방문자 수", sm.totalVisitors || 0, new Date()]);
    summarySheet.appendRow(["총 작업 횟수", sm.totalAllFeaturesUsed || 0, new Date()]);
    summarySheet.appendRow(["PDF 병합 횟수", payload.analytics?.featureUsage?.pdfMerge?.totalExecutionCount || 0, new Date()]);
    summarySheet.appendRow(["Word 변환 횟수", payload.analytics?.featureUsage?.wordConvert?.totalExecutionCount || 0, new Date()]);
    summarySheet.appendRow(["PPT 변환 횟수", payload.analytics?.featureUsage?.pptConvert?.totalExecutionCount || 0, new Date()]);

    // 2. 방문자 로그 시트
    let visitorSheet = ss.getSheetByName("방문자현황");
    if (!visitorSheet) {
      visitorSheet = ss.insertSheet("방문자현황");
      visitorSheet.appendRow(["방문자ID", "이메일/계정", "기기/브라우저", "방문횟수", "작업횟수", "최근접속일시"]);
    }
    if (payload.visitors && Array.isArray(payload.visitors)) {
      visitorSheet.clearContents();
      visitorSheet.appendRow(["방문자ID", "이메일/계정", "기기/브라우저", "방문횟수", "작업횟수", "최근접속일시"]);
      payload.visitors.forEach(v => {
        visitorSheet.appendRow([v.visitorId, v.email || "익명", v.userAgent, v.visitCount, v.actionsCount, v.lastVisit]);
      });
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Data synced successfully",
      syncedAt: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  const key = e.parameter ? e.parameter.apiKey : "";
  if (key !== VALID_API_KEY) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Invalid API Key" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    ping: "pong",
    time: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}
